
getStackQualifiedName(lookupID: string): 

  All stack references are qualified with the environment in which they are declared.
  
  Algorithm:
    1. cEnv = findContainingEnvironment(lookupID)
    2. return "ENV" + cEnv.idx + "$" + lookupID 

  Sample Input: 
    import { default as boo } from "some.js";

  Sample Output:
    E1$boo

getHeapQualifiedName(lookupID: string):

  All heap references, similar to stack references are qualified with a specific naming scheme.

  ## WARNING ## : Heap Objects are qualfied by their declaring BBs and their offsets inside those basic blocks

  Algorithm:
    1. dBB = declaringBB()
    2. sOffset = instructionOffsetInside(dBB)
    3. return "BB" + dBB.idx + "$" + sOffset 


getPointees(lookupID: string):
  
  Returns the set of nodes pointed by a stack object

  Algorithm:
    1. ensureNode(lookupID)
    2. succ = this.getSuccessors(lookupID)
    3. return succ.map(s => nodeMap.get(s))


dissernPointees(objs: Array<PTANode>):

  If an object points to a "knowable" field object, then we can dissern what symbol it is.

  Algorithm:
    1. let res = Set()
    2. for (o of objs) 
      If: res is a dissernable node, update res with value 
      Else: add "*" generic case to res
    3. return res


PTAGraph Properties:
  
  At each point, we maintain a graph data structure for storing the PTA.
  Each node in this graph is associated with an object of class PTANode.
  We name this mapping as "nodeMap : Map<string, PTANode>".





PTA Transfer Function(s): 

-- BB() {
  let inState = resolveState()
  let nextState = inState.clone()

  for (s of this.statements) {

    -- IS_Break | IS_LBreak | IS_Continue | IS_LContinue | IS_Debugger | IS_Return | IS_Throw | IS_AImport --> NOOP

    -- IS_BImport {
      stackID = getStackQualifiedName(s.local)
      stackNode = new StackNode(stackID)
      nextState.declareNode(stackNode)

      heapID  = getStackQualifiedName(s.local)
      heapNode = new LazyHeapNode()
      nextState.declareNode(heapNode)

      nextState.clearSuccessors(stackNode)
      nextState.drawStackToHeapEdge(stackNode, heapNode)
    }

    -- IS_ClassStaticPropInit {
      receiverObjID = getStackQualifiedName(s.obj)
      nextState.ensureNode(receiverObjID)

      rValID = getStackQualifiedName(i.RVal.lookupName())
      nextState.ensureNode(rValID)
      
      receiverPointees = nextState.getPointees(receiverObjID)
      rValPointees = nextState.getPointees(rValID)

      dissernedProps : set<string>
      if (s.computedProp()) {
        propID = getStackQualifiedName(s.propID)
        nextState.ensureNode(propID)
        propPointees = nextState.getPointees(propID)
        dissernedProps = dissernPointees(propPointees)
      } else {
        dissernedProps = [propID.name]
      }

      nextState.drawHeapToHeapEdge(receiverPointees, rValPointees, dissernedProps)
    }

    -- IS1_DeclarationStmt {
      Pattern matching on LVAL destructuring pattern
      Generate RVALS
    }

    -- IS1_AssignmentStmt {
      Pattern matching on LVAL destructuring pattern
      Generate RVALS
    }

  }
}
